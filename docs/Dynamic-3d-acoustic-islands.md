```
# Minimal example: take an audio file, break into segments, produce random demo points per timeframe.
# Replace with your real per-segment embeddings -> UMAP(3) -> timeline export.
import json, numpy as np

# create demo timeline: 200 points, 120 timesteps
num_points = 200
timesteps = 120
points_timeline = []
np.random.seed(42)
base_positions = np.random.normal(scale=3.0, size=(num_points,3))
# create clusters by offsetting sets of points
for t in range(timesteps):
    frame = []
    jitter = np.random.normal(scale=0.08, size=(num_points,3))
    # move cluster centers slowly
    phase = np.sin(np.linspace(0,2*np.pi, num_points) + t*0.05)
    positions = base_positions + (phase[:,None]*0.3) + jitter
    for i,pos in enumerate(positions):
        cluster = int((i/num_points)*5)
        energy = float(abs(phase[i]))
        frame.append({"id": i, "x": float(pos[0]), "y": float(pos[1]), "z": float(pos[2]), "cluster": cluster, "energy": energy})
    points_timeline.append(frame)

with open('points_timeline.json','w') as f:
    json.dump({"timesteps": timesteps, "points": points_timeline}, f)
print('wrote points_timeline.json')
```

import React, { useRef, useEffect, useState, Suspense } from 'react'
import { Canvas, useFrame, useThree } from '@react-three/fiber'
import { OrbitControls, Html } from '@react-three/drei'
import * as THREE from 'three'
import create from 'zustand'

// simple store for timeline playback control
const useStore = create(set => ({
  timeIndex: 0,
  playing: true,
  setTime: (t) => set({ timeIndex: t }),
  setPlaying: (p) => set({ playing: p }),
}))

function useTimeline(url) {
  const [data, setData] = useState(null)
  useEffect(() => {
    let mounted = true
    fetch(url)
      .then(r => r.json())
      .then(j => { if (mounted) setData(j) })
      .catch(e => console.error('Failed load timeline', e))
    return () => { mounted = false }
  }, [url])
  return data
}

// PointsCloud component: handles GPU BufferGeometry and animates between frames
function PointsCloud({ timeline, pointSize = 4 }) {
  const ref = useRef()
  const meshRef = useRef()
  const { camera } = useThree()
  const [current, setCurrent] = useState(0)
  const [next, setNext] = useState(1)
  const playing = useStore(state => state.playing)
  const timeIndex = useStore(state => state.timeIndex)

  // initialize positions from first frame
  useEffect(() => {
    if (!timeline) return
    const points = timeline.points[0]
    const num = points.length
    // create buffers
    const positions = new Float32Array(num * 3)
    const targets = new Float32Array(num * 3)
    const colors = new Float32Array(num * 3)
    const sizes = new Float32Array(num)
    for (let i=0;i<num;i++){
      const p = points[i]
      positions[i*3+0] = p.x
      positions[i*3+1] = p.y
      positions[i*3+2] = p.z
      targets[i*3+0] = p.x
      targets[i*3+1] = p.y
      targets[i*3+2] = p.z
      // color by cluster
      const c = clusterColor(p.cluster)
      colors[i*3+0] = c.r
      colors[i*3+1] = c.g
      colors[i*3+2] = c.b
      sizes[i] = 1 + p.energy*2
    }
    // attach attributes
    const geometry = new THREE.BufferGeometry()
    geometry.setAttribute('position', new THREE.BufferAttribute(positions,3))
    geometry.setAttribute('target', new THREE.BufferAttribute(targets,3))
    geometry.setAttribute('color', new THREE.BufferAttribute(colors,3))
    geometry.setAttribute('size', new THREE.BufferAttribute(sizes,1))
    if (ref.current) ref.current.geometry = geometry
  }, [timeline])

  // keep internal phase for interpolation
  const interpRef = useRef({ alpha: 0, duration: 0.15, lastT: 0 })

  useFrame((state, delta) => {
    if (!timeline || !ref.current) return
    const storeTime = useStore.getState().timeIndex
    // if the global timeIndex changed, update target buffer to next frame
    if (storeTime !== interpRef.current.lastT) {
      const t = storeTime % timeline.timesteps
      const t2 = (t+1) % timeline.timesteps
      interpRef.current.lastT = t
      interpRef.current.alpha = 0
      interpRef.current.duration = Math.max(0.08, 0.12) // per-frame lerp time, tuneable
      // update 'target' attribute with frame t2
      const targets = ref.current.geometry.getAttribute('target')
      const pts = timeline.points[t2]
      for (let i=0;i<pts.length;i++){
        targets.array[i*3+0] = pts[i].x
        targets.array[i*3+1] = pts[i].y
        targets.array[i*3+2] = pts[i].z
      }
      targets.needsUpdate = true
    }

    // animate interpolation between position -> target
    interpRef.current.alpha += delta / interpRef.current.duration
    const a = Math.min(1, interpRef.current.alpha)
    const pos = ref.current.geometry.getAttribute('position')
    const targ = ref.current.geometry.getAttribute('target')
    const colors = ref.current.geometry.getAttribute('color')
    const sizes = ref.current.geometry.getAttribute('size')
    // smooth movement: lerp each vertex
    for (let i=0;i<pos.count;i++){
      const ix = i*3
      // apply easing (smoothstep)
      const e = easeInOutCubic(a)
      pos.array[ix+0] = THREE.MathUtils.lerp(pos.array[ix+0], targ.array[ix+0], e*0.98)
      pos.array[ix+1] = THREE.MathUtils.lerp(pos.array[ix+1], targ.array[ix+1], e*0.98)
      pos.array[ix+2] = THREE.MathUtils.lerp(pos.array[ix+2], targ.array[ix+2], e*0.98)
      // optional tiny breathing motion to make it feel alive
      const breathe = Math.sin((state.clock.elapsedTime + i*0.01))*0.002
      pos.array[ix+1] += breathe * (sizes.array[i] || 1.0)
    }
    pos.needsUpdate = true

    // update a rotation or attraction field for nicer visuals
    if (meshRef.current) meshRef.current.rotation.y += delta*0.02
  })

  if (!timeline) return null
  const num = timeline.points[0].length
  const material = (
    <pointsMaterial attach="material" vertexColors size={pointSize} sizeAttenuation />
  )

  return (
    <points ref={meshRef}>
      <bufferGeometry ref={ref} />
      {material}
    </points>
  )
}

// helper: color palette for clusters
function clusterColor(cluster) {
  const palette = [
    new THREE.Color('#2b6cb0'),
    new THREE.Color('#38a169'),
    new THREE.Color('#d69e2e'),
    new THREE.Color('#9f7aea'),
    new THREE.Color('#e53e3e'),
    new THREE.Color('#ed64a6'),
  ]
  return palette[cluster % palette.length]
}

// small easing
function easeInOutCubic(t){ return t<0.5 ? 4*t*t*t : 1 - Math.pow(-2*t+2,3)/2 }

// UI controls component
function Controls({ timeline }){
  const playing = useStore(state => state.playing)
  const setPlaying = useStore(state => state.setPlaying)
  const timeIndex = useStore(state => state.timeIndex)
  const setTime = useStore(state => state.setTime)

  useEffect(()=>{
    let raf
    function loop(){
      if (useStore.getState().playing && timeline){
        const t = (useStore.getState().timeIndex + 1) % timeline.timesteps
        useStore.getState().setTime(t)
      }
      raf = setTimeout(loop, 120) // advance every 120ms; tune to match your audio timeframe
    }
    loop()
    return ()=> clearTimeout(raf)
  }, [timeline])

  return (
    <div className="absolute left-4 bottom-4 bg-black/40 text-white p-3 rounded-md backdrop-blur-sm">
      <div className="flex gap-2 items-center">
        <button onClick={() => setPlaying(!playing)} className="px-3 py-1 rounded bg-white/10">{playing? 'Pause':'Play'}</button>
        <input type="range" min={0} max={timeline?timeline.timesteps-1:0} value={timeIndex} onChange={(e)=>setTime(Number(e.target.value))} />
        <div className="text-xs">Frame: {timeIndex}/{timeline?timeline.timesteps-1:0}</div>
      </div>
    </div>
  )
}

export default function App(){
  // expecting the points_timeline.json to be in public folder or served
  const timeline = useTimeline('/points_timeline.json')
  return (
    <div className="w-screen h-screen bg-gradient-to-br from-gray-900 to-black">
      <Canvas camera={{ position: [0,0,12], fov: 50 }}>
        <ambientLight intensity={0.6} />
        <pointLight position={[10,10,10]} intensity={0.8} />
        <Suspense fallback={null}>
          {timeline && <PointsCloud timeline={timeline} pointSize={4} />}
        </Suspense>
        <OrbitControls enablePan={true} enableZoom={true} enableRotate={true} />
      </Canvas>
      {timeline ? <Controls timeline={timeline} /> : <div className="absolute left-4 bottom-4 text-white p-2">Loading timeline...</div>}
    </div>
  )
}

- Use temporal smoothing and per-point oscillators (small offsets driven by per-point "energy" or cluster phase).
- Use an attractive force: points slightly attract towards cluster centroids while keeping noise to create organic islands.
- Use additive blending + small glow (postprocessing) to get soft, luminous islands.
- Drive the timeline by audio envelope or segmentation timestamps so visual motion aligns with audio.
- Add lines between temporally adjacent points of the same cluster to show motifs.
